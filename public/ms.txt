# 1. Clean up some disk space first (to avoid 'No space left on device' errors)
sudo apt-get clean
sudo journalctl --vacuum-time=1d
sudo rm -rf /var/cache/apt/archives/*

# 2. Add the necessary repositories (contrib and non-free) for the font installer
sudo apt-get update
sudo apt-get install -y software-properties-common
sudo apt-add-repository -y contrib
sudo apt-add-repository -y non-free

# 3. Accept the Microsoft End User License Agreement (EULA) automatically
echo ttf-mscorefonts-installer msttcorefonts/accepted-mscorefonts-eula select true | sudo debconf-set-selections

# 4. Install the fonts and fontconfig in non-interactive mode
sudo DEBIAN_FRONTEND=noninteractive apt-get update
sudo DEBIAN_FRONTEND=noninteractive apt-get install -y \
    ttf-mscorefonts-installer \
    fonts-crosextra-carlito \
    fonts-crosextra-caladea \
    fontconfig

# 5. Refresh the system's font cache so LibreOffice can detect them immediately
sudo fc-cache -f -v
